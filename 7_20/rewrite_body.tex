\begin{frame}{LatentMAS: latent thought와 KV working memory}
\small
\begin{columns}[T,onlytextwidth]
\column{0.47\textwidth}
\begin{block}{Latent thought generation}
\[
h_t = f_\theta(e_{\le t}), \qquad
e_{t+1}=h_t W_a
\]
\[
W_a \approx W_{\mathrm{out}}^\dagger W_{\mathrm{in}}
\]
\scriptsize
token을 decode하지 않고 last-layer hidden state를 다음 input embedding으로 다시 넣는다.
\end{block}

\column{0.49\textwidth}
\begin{block}{KV working memory transfer}
\[
M_A=\{(K_A^\ell,V_A^\ell)\}_{\ell=1}^{L}
\]
\[
K_B^\ell \leftarrow [K_A^\ell;K_B^\ell], \quad
V_B^\ell \leftarrow [V_A^\ell;V_B^\ell]
\]
\scriptsize
agent A가 만든 layer-wise KV cache를 agent B의 prefix memory처럼 붙인다.
\end{block}
\end{columns}

\begin{frame}{LatentMAS의 한계: heterogeneous setting}
\small
\begin{block}{논문 구조의 강한 가정}
\begin{itemize}
  \item 같은 모델 family 또는 같은 shape의 layer-wise KV를 전제로 하기 쉽다.
  \item 서로 다른 layer 수, KV head 수, head dimension에 대한 변환은 main method가 아니다.
  \item heterogeneous extension은 adapter가 필요하다는 future direction에 가깝다.
\end{itemize}
\end{block}

\begin{frame}{RecursiveMAS: inner-loop와 outer communication}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Inner recursive link}
\[
R_{\mathrm{in}}(h)=h+W_2\sigma(W_1h)
\]
\[
\mathcal{L}_{\mathrm{in}}
=1-\cos(R_{\mathrm{in}}(H), \mathrm{Emb}(y))
\]
\scriptsize
latent state를 정답 label 또는 teacher-generated target 쪽으로 정렬한다.
\end{block}

\column{0.48\textwidth}
\begin{block}{Outer communication}
\[
R_{\mathrm{out}}(h)=W_3h+W_2\sigma(W_1h)
\]
\scriptsize
agent 사이에는 주로 last-layer hidden representation을 adapter로 변환해 전달한다.
\end{block}
\end{columns}

\begin{frame}{RecursiveMAS의 약점}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{학습 supervision 문제}
\begin{itemize}
  \item closed-form QA에서는 정답 label이 명확하다.
  \item planner/refiner target은 teacher rewrite에 의존한다.
  \item open-ended MAS에서는 role-level gold trace가 애매하다.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{전달 단위 문제}
\begin{itemize}
  \item layer-wise KV가 아니라 hidden representation 전달이다.
  \item transformer attention memory 자체를 넘기지 않는다.
  \item heterogeneous agent 간 shape mismatch를 별도 처리해야 한다.
\end{itemize}
\end{block}
\end{columns}

\vspace{0.8em}
\begin{block}{}
\small
우리 실험에서는 RecursiveMAS inner-loop는 비교 공정성을 위해 유지하고, cross-agent communication만 KV prefix translation으로 교체한다.
\end{block}
\end{frame}

\begin{frame}{Cache-to-Cache: pairwise cache fusion}
\small
\begin{block}{C2C의 핵심 수식}
\[
C_F^n
=C_R^n(X)+F_n\!\left(C_R^n(X),\,C_S^{G(n)}(X)\right)
\]
\[
y_{i+1}=P\!\left(y_i \mid C_F(X)\oplus C(Y_{0:i})\right)
\]
\end{block}

\scriptsize
C2C는 receiver가 이미 같은 input \(X\)를 prefill해서 만든 cache를 가지고 있고, 여기에 sharer cache를 residual하게 섞는 구조다.
\end{frame}

\begin{frame}{C2C에서 fusion을 쓰는 이유}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{왜 receiver KV를 같이 보나?}
\begin{itemize}
  \item Sharer와 Receiver가 같은 context \(X\)를 인코딩한다.
  \item Receiver의 기존 이해를 보존해야 한다.
  \item Gate가 도움이 되는 layer만 선택한다.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{우리 설정과 다른 점}
\begin{itemize}
  \item agent마다 role prompt와 objective가 다르다.
  \item Receiver KV를 만들려면 receiver prefill을 먼저 해야 한다.
  \item 우리는 receiver prefill 전에 prefix memory를 주입하고 싶다.
\end{itemize}
\end{block}
\end{columns}

\vspace{0.7em}
\begin{block}{}
\small
따라서 HeteroMAS는 C2C-style fusion이 아니라 \textbf{sender-only KV translation + prefix prepend}를 기본 구조로 둔다.
\end{block}
\end{frame}

\end{tikzpicture}
\end{center}

\vspace{0.5em}
\begin{block}{실행 순서}
\[
A\ \mathrm{forward}
\rightarrow C_A
\rightarrow \hat{C}_{A\rightarrow B}
\rightarrow B\ \mathrm{forward}(\mathrm{past\_key\_values}=\hat{C}_{A\rightarrow B})
\]
\end{block}

\vspace{0.3em}
\begin{block}{}
\scriptsize
중요: receiver KV를 먼저 만들고 섞는 것이 아니라, receiver가 자기 prompt를 처리하기 \textbf{전에} translated KV가 prefix로 들어간다.
\end{block}
\end{frame}

\begin{frame}{왜 receiver KV를 안 보나?}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Receiver KV를 보면}
\begin{itemize}
  \item Receiver prompt prefill이 먼저 필요하다.
  \item 이후 cache editing/fusion 문제가 된다.
  \item text 없이도 두 모델의 cache를 token-level로 맞춰야 한다.
\end{itemize}
\end{block}

\column{0.48\textwidth}
\begin{block}{Sender-only translation이면}
\begin{itemize}
  \item sender의 latent memory만 receiver shape으로 바꾸면 된다.
  \item token length alignment가 필요 없다.
  \item receiver는 prefix를 보고 자기 role reasoning을 시작한다.
\end{itemize}
\end{block}
\end{columns}

\begin{frame}{Layer alignment: depth-normalized mapping}
\small
\begin{block}{Receiver-driven mapping}
Receiver layer \(r\in\{0,\ldots,L_B-1\}\)에 대해 sender layer를 선택한다.
\[
s(r)
=\mathrm{round}\!\left(
r\cdot \frac{L_A-1}{L_B-1}
\right)
\]
\[
\ell_A=s(r), \qquad \ell_B=r
\]
\end{block}

\begin{frame}{CacheTranslator: token-wise MLP projection}
\small
\begin{block}{Sender KV}
\[
K_A^{s(r)},V_A^{s(r)}
\in \mathbb{R}^{B\times H_A\times T_A\times d_A}
\]
\end{block}

\begin{block}{Head와 head dimension만 flatten, token 축은 유지}
\[
X_A^r
=\Flatten([K_A^{s(r)},V_A^{s(r)}])
\in \mathbb{R}^{B\times T_A\times 2H_A d_A}
\]
\[
Y_A^r
=W_2\sigma(W_1X_A^r + e_r)
\in \mathbb{R}^{B\times T_A\times 2H_B d_B}
\]
\end{block}

\begin{block}{Receiver shape으로 복원}
\[
(\hat K_{A\rightarrow B}^{r},\hat V_{A\rightarrow B}^{r})
=\Reshape_B(Y_A^r)
\in \mathbb{R}^{B\times H_B\times T_A\times d_B}
\]
\end{block}
\end{frame}

\begin{frame}{Receiver attention: prepend 방식}
\small
\begin{block}{Translated prefix KV}
\[
\hat{C}_{A\rightarrow B}
=\{(\hat K_{A\rightarrow B}^{r},\hat V_{A\rightarrow B}^{r})\}_{r=0}^{L_B-1}
\]
\end{block}

\begin{block}{Receiver가 자기 prompt를 처리할 때}
\[
K_{\mathrm{total}}^r=[\hat K_{A\rightarrow B}^{r};K_B^r],
\qquad
V_{\mathrm{total}}^r=[\hat V_{A\rightarrow B}^{r};V_B^r]
\]
\[
\mathrm{Attn}(Q_B^r,K_{\mathrm{total}}^r,V_{\mathrm{total}}^r)
=
\mathrm{softmax}\!\left(
\frac{Q_B^r(K_{\mathrm{total}}^r)^\top}{\sqrt{d_B}}
\right)V_{\mathrm{total}}^r
\]
\end{block}

\begin{block}{}
\scriptsize
토큰 길이 \(T_A\)를 \(T_B\)에 맞출 필요가 없다. Prefix sequence가 앞에 추가될 뿐이고, receiver의 실제 prompt token은 그대로 유지된다.
\end{block}
\end{frame}

\begin{frame}{Multi-agent 구조에서의 KV 전달}
\small
\begin{columns}[T,onlytextwidth]
\column{0.48\textwidth}
\begin{block}{Sequential}
\[
\mathrm{Planner}
\rightarrow \hat{C}_{P\rightarrow R}
\rightarrow \mathrm{Refiner}
\rightarrow \hat{C}_{R\rightarrow S}
\rightarrow \mathrm{Solver}
\]
\scriptsize
Recursive round가 있으면 solver KV를 다음 round planner prefix로 번역한다.
\end{block}

\column{0.48\textwidth}
\begin{block}{Hierarchical}
\[
\hat C_{M\rightarrow Z}
\oplus
\hat C_{C\rightarrow Z}
\oplus
\hat C_{S\rightarrow Z}
\rightarrow
\mathrm{Summarizer}
\]
\scriptsize
각 expert KV를 summarizer-shaped KV로 번역하고 sequence dimension으로 concat한다.
\end{block}
\end{columns}

\begin{frame}{정리}
\small
\begin{enumerate}
  \item LatentMAS는 text decoding을 줄이지만 homogeneous KV working memory 가정이 강하다.
  \item RecursiveMAS는 heterogeneous MAS 구조를 갖지만 hidden-state 전달과 role-level supervision 부담이 있다.
  \item C2C는 KV의 semantic transfer 가능성을 보였지만 receiver KV와의 fusion이 핵심이다.
  \item HeteroMAS는 RecursiveMAS의 비교 구조를 유지하되, cross-agent 전달을 hidden state가 아니라 KV prefix translation으로 바꾼다.
\end{enumerate}
